% 两杆角均是相对竖直的绝对角；慢模式同相、快模式反相，下球振幅由耦合矩阵给出而非任意指定。
\begin{tikzpicture}[scale=.95]
\foreach \shift/\sgn/\labeltext in {0/1/慢模式,5.5/-1/快模式}{
\begin{scope}[xshift=\shift cm]
\pgfmathsetmacro{\angtwo}{\sgn*sqrt(2)*12}
\coordinate(O)at(0,0);
\coordinate(A)at({2*sin(12)},{-2*cos(12)});
\coordinate(B)at($(A)+({2*sin(\angtwo)},{-2*cos(\angtwo)})$);
\draw[thick](-.7,0)--(.7,0);
\draw[guide](O)--(0,-1.65);\draw[guide](A)--++(0,-1.65);
\draw[ink,thick](O)--(A)node[midway,right=4pt]{$l$}--(B);\ifnum\sgn=1\node[right=4pt]at($(A)!.3!(B)$){$l$};\else\node[left=4pt]at($(A)!.3!(B)$){$l$};\fi
\fill(A)circle(4pt)node[right=4pt]{$m$};\fill(B)circle(4pt)node[right=4pt]{$m$};
\draw(0,-.78)arc(-90:-78:.78);\node[left]at(-.04,-1){$\theta_1$};
\draw($(A)+(0,-.78)$)arc(-90:{-90+\angtwo}:.78);
\ifnum\sgn=1\node[left]at($(A)+(-.08,-1.1)$){$\theta_2$};
\else\node[right]at($(A)+(.08,-1.1)$){$\theta_2$};\fi
\node at(.3,-4.5){\labeltext};
\ifnum\sgn=1\node at(.3,-5.05){$\Theta_2=\sqrt2\Theta_1$};
\else\node at(.3,-5.05){$\Theta_2=-\sqrt2\Theta_1$};\fi
\end{scope}}
\end{tikzpicture}
